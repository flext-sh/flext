#### ✅ CONFIGURATION-DRIVEN (aplicando REGRA #4)

```python
class ConfigurationDrivenProcessor:
    def __init__(self, rules_path: str):
        self.rules = self._load_rules(rules_path)

    def process_acl(self, entry: dict) -> str:
        """REGRA #4: Configuration_Driven_Processing"""

        # Permissões baseadas em contexto (não hardcoded)
        context = self._determine_context(entry["dn"])
        permission_mapping = self.rules["transformation_definitions"]["convert_orclaci_to_oud"]["permission_mapping"]

        # Template configurável (não hardcoded)
        template = self.rules["transformation_definitions"]["convert_orclaci_to_oud"]["template"]

        # Subject mapping configurável
        subject_mapping = self.rules["transformation_definitions"]["convert_orclaci_to_oud"]["subject_mapping"]

        # Processing baseado em configuração
        permissions = self._map_permissions(entry["permissions"], permission_mapping, context)
        bind_rule = self._map_subject(entry["subject"], subject_mapping)

        return template.format(
            dn=entry["dn"],
            oud_permissions=",".join(permissions),
            bind_rule=bind_rule,
            dn_simple=self._clean_dn_for_name(entry["dn"])
        )

    def _determine_context(self, dn: str) -> str:
        """Determina contexto baseado no DN"""
        if "Groups" in dn:
            return "group_context"
        elif "Users" in dn:
            return "user_context"
        elif "OracleContext" in dn:
            return "admin_context"
        return "default_context"
```

```python
def apply_consolidation_rule(entries: list) -> list:
    """REGRA #3: Entry_Consolidation_Optimization"""

    # Step 1: Group by target DN
    groups = {}
    for entry in entries:
        target_dn = extract_target_dn(entry)
        if target_dn not in groups:
            groups[target_dn] = []
        groups[target_dn].append(entry)

    # Step 2: Consolidate multiple ACIs per target
    consolidated = []
    for target_dn, group_entries in groups.items():
        if len(group_entries) == 1:
            # Single ACI - convert to modify format
            consolidated.append(convert_to_modify_format(group_entries[0]))
        else:
            # Multiple ACIs - consolidate into single modify entry
            consolidated.append(create_multi_aci_modify_entry(target_dn, group_entries))

    # Step 3: Sort by hierarchy and dependencies
    sorted_entries = sort_by_dn_depth(consolidated)

    return sorted_entries

def create_multi_aci_modify_entry(target_dn: str, entries: list) -> dict:
    """Create single modify entry with multiple ACI additions"""
    modify_entry = {
        "dn": target_dn,
        "changetype": "modify"
    }

    # Add each ACI as separate add operation
    for i, entry in enumerate(entries):
        if i == 0:
            modify_entry["add"] = "aci"
            modify_entry["aci"] = entry["aci"]
        else:
            modify_entry[f"add_{i}"] = "aci"
            modify_entry[f"aci_{i}"] = entry["aci"]

        # Add separator between operations
        if i < len(entries) - 1:
            modify_entry[f"separator_{i}"] = "-"

    return modify_entry
```
