    ```python
    class FlextInfra(s[bool]):
        """Coordinate infrastructure operations via factory-method accessors.

        Domain services have incompatible type parameters (s[bool] vs s[str]),
        making MRO composition infeasible. Instead, each domain is accessed
        via a factory method that returns the domain's service instance.
        """

        _instance: ClassVar[Self | None] = None

        @classmethod
        def get_instance(cls) -> Self:
            if cls._instance is None:
                cls._instance = cls()
            return cls._instance

        @override
        def execute(self) -> r[bool]:
            return r[bool].ok(True)

        # Factory-method accessors for each domain
        @staticmethod
        def basemk(**kwargs: ...) -> FlextInfraBaseMkGenerator:
            return FlextInfraBaseMkGenerator(**kwargs)

        @staticmethod
        def check(**kwargs: ...) -> FlextInfraWorkspaceChecker:
            return FlextInfraWorkspaceChecker(**kwargs)

        # ... etc for each domain
    ```
